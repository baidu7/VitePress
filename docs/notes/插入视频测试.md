<script setup>
import { onMounted, onBeforeUnmount } from 'vue'

// 这里的变量要定义在外面，方便清理函数调用
let startPrank = null

onMounted(() => {
  const v = document.querySelector('.custom-video')
  
  startPrank = () => {
    // 关键：先确认视频元素是否还在页面上，且没有被隐藏
    if (v && document.body.contains(v)) {
      v.muted = false 
      v.volume = 1.0 
      
      const playPromise = v.play()

      if (playPromise !== undefined) {
        playPromise.then(() => {
          console.log("雷击成功！")
          // 成功后移除当前页面的监听
          document.removeEventListener('click', startPrank)
        }).catch(error => {
          console.log("浏览器还在抵抗: ", error)
        })
      }
    } else {
      // 如果找不到视频，说明已经不在该文章页了，自动清理监听器
      document.removeEventListener('click', startPrank)
    }
  }

  // 监听全页面的点击
  document.addEventListener('click', startPrank)
})

// --- 核心：离开页面时必须撤销监听 ---
onBeforeUnmount(() => {
  if (startPrank) {
    document.removeEventListener('click', startPrank)
    console.log("已清理整蛊监听，放过下一位读者")
  }
})
</script>
